1. tap(x, y) 【点击】
   功能: 在逻辑坐标 (x,y) 单击。横屏 init(1/2) 时坐标为逻辑横屏画布
   参数: x(number) - 逻辑X, y(number) - 逻辑Y
   返回: 无返回值
   实例: init(1)
         local x, y = 100, 200
         if x > 0 then
           tap(x, y)
           mSleep(300)
         end

2. swipe(x1, y1, x2, y2, ms) 【滑动】
   功能: 逻辑坐标滑动，时长毫秒
   参数: x1,y1(number) - 起点, x2,y2(number) - 终点, ms(number) - 时长毫秒
   返回: 无返回值
   实例: local x1, y1, x2, y2 = 80, 120, 200, 120
         local dur = 250
         if dur > 0 then swipe(x1, y1, x2, y2, dur) end

3. keepScreen(on) 【锁帧缓存】
   功能: 开启后找色优先复用缓存帧，停脚本自动释放
   参数: on(boolean) - true开启/false关闭
   返回: boolean - 是否成功
   实例: keepScreen(true)
         for i = 1, 5 do
           local c = getColor(10, 10)
           if c >= 0 then break end
           mSleep(200)
         end
         keepScreen(false)

4. getColor(x, y) 【取色】
   功能: 读取逻辑坐标颜色，返回 0xRRGGBB 整数；失败可能为 -1
   参数: x,y(number) - 逻辑坐标
   返回: number - 0xRRGGBB 或 -1
   实例: local cx, cy = 10, 10
         local c = getColor(cx, cy)
         if type(c) == "number" and c >= 0 then
           toast(string.format("c=0x%06x", c % 0x1000000), 800)
         end

5. findMultiColorInRegionFuzzy(main, offsetStr, degree, x1, y1, x2, y2) 【多点模糊找色】
   功能: 多点组合模糊找色，degree∈[1,100]，建议局部区域+keepScreen
   参数: main(number) - 主色0xRRGGBB, offsetStr(string) - dx|dy|0xRGB,..., degree(number) - 相似度1-100, x1,y1,x2,y2(number) - 区域
   返回: number,number - 命中逻辑坐标；未命中 -1,-1
   实例: keepScreen(true)
         local main = 0xfdfeeb
         local off = "0|1|0xfdfef8,0|2|0xfffff7"
         local x, y = findMultiColorInRegionFuzzy(main, off, 90, 2012, 281, 2012, 284)
         if x ~= -1 then
           tap(x, y)
         else
           toast("searching", 500)
         end
         keepScreen(false)

6. toast(text, ms) 【底部提示】
   功能: HUD提示，横竖屏统一逻辑底边居中
   参数: text(string) - 文本, ms(number) - 显示毫秒(可选)
   返回: 无返回值
   实例: local msg = "子砚Toast"
         local ms = 1200
         if #msg > 0 then toast(msg, ms) end

7. mSleep(ms) 【延时】
   功能: 毫秒延时；内含暂停/停止检查点
   参数: ms(number) - 毫秒
   返回: 无返回值
   实例: for i = 1, 3 do
           mSleep(200)
         end

8. init(orient) 【初始化方向】
   功能: 0竖屏跟随/1横屏CCW/2横屏CW，影响逻辑坐标与Toast/找色
   参数: orient(number) - 0/1/2
   返回: 无返回值
   实例: local orient = 1

         if orient == 1 or orient == 2 then
           toast("landscape", 800)
         end

9. snapshot(path) 【截屏存盘】
   功能: 将当前逻辑屏截图保存到path（png）
   参数: path(string) - 输出路径
   返回: 无/依赖实现

init(1)
keepScreen(true)
for i = 1, 10 do
  local c = getColor(120, 120)
  if type(c) == "number" and c >= 0 then
    local x, y = findMultiColorInRegionFuzzy(c, string.format("0|0|0x%06x", c % 0x1000000), 85, 110, 110, 130, 130)
    if x ~= -1 then
      tap(x, y)
      break
    end
  end
  mSleep(200)
end
keepScreen(false)

  local suggest = Zy.StateMachine.suggest(currentState)

  end

end
Zy.Screen.keep(false)

================================================================================
函数使用注意事项
================================================================================

1. 坐标系统
   - 全局坐标基于逻辑坐标，不是物理像素
   - 建议使用 init(1) 或 init(2) 初始化方向
   - keepScreen(true) 可显著提升找色性能

2. 性能优化
   - 找色前务必开启 keepScreen(true)
   - 使用局部区域找色，避免全屏扫描
   - OCR识别间隔建议 200-350ms
   - 找色相似度建议 85-95

3. 错误处理
   - getColor 返回 -1 表示取色失败
   - findMultiColorInRegionFuzzy 返回 -1,-1 表示未命中
   - OCR 返回 false 表示识别失败

4. 多机型适配测试
   - 支持机型: iPhone 7 (A1660/A1778)、iPhone 7 Plus (A1661/A1784)、iPhone 8 (A1863/A1905)、iPhone 8 Plus (A1864/A1898)
   - 系统版本: iOS 13.x - iOS 16.x
   - 越狱环境: rootless (Dopamine/Unc0ver) 与 rootful (Pangu/PP)
   - 禁止单设备坐标特判
   - 所有修复必须多机型通用
   - 分辨率适配: 统一使用 ScreenTransform 逻辑坐标系，禁止硬编码物理像素
   - 内存管理: 遵循 keepScreen 缓存机制，30秒缓存TTL
   - iOS 16+ 适配: 使用 UIGraphicsImageRenderer 替代 _UICreateScreenUIImage 截屏
   - A11 芯片优化: NCNN 模型 int8 量化，CPU 推理，禁止 ANE/MLX 硬件加速
