## rootless runtime-root selection
     1	#!/bin/sh
     2	# Source-only runtime scheme resolver. postinst writes the marker from the
     3	# installed deb Architecture; `/var/jb` may exist on a rootful device.
     4	
     5	ZIYAN_RUNTIME_SCHEME_MARKER="/var/mobile/Library/Preferences/com.ziyan.ziyan.runtime_scheme"
     6	ZIYAN_RUNTIME_SCHEME=""
     7	if [ -r "$ZIYAN_RUNTIME_SCHEME_MARKER" ]; then
     8	  IFS= read -r ZIYAN_RUNTIME_SCHEME <"$ZIYAN_RUNTIME_SCHEME_MARKER" || true
     9	fi
    10	
    11	case "$ZIYAN_RUNTIME_SCHEME" in
    12	  rootless) ZIYAN_RUNTIME_ROOT="/var/jb/usr/lib/ziyan" ;;
    13	  rootful) ZIYAN_RUNTIME_ROOT="/usr/lib/ziyan" ;;
    14	  *)
    15	    # Legacy fallback before a marker exists. A newly installed package always
    16	    # has one, so the ambiguous dual-tree case never reaches this branch.
    17	    if [ -f /var/jb/usr/lib/ziyan/lib/lua/ziyan_run.lua ] &&
    18	       [ ! -f /usr/lib/ziyan/lib/lua/ziyan_run.lua ]; then
    19	      ZIYAN_RUNTIME_SCHEME="rootless"
    20	      ZIYAN_RUNTIME_ROOT="/var/jb/usr/lib/ziyan"
    21	    else
    22	      ZIYAN_RUNTIME_SCHEME="rootful"
    23	      ZIYAN_RUNTIME_ROOT="/usr/lib/ziyan"
    24	    fi
    25	    ;;
    26	esac
    27	export ZIYAN_RUNTIME_SCHEME ZIYAN_RUNTIME_ROOT
